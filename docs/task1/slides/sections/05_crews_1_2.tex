\begin{block}{Crew 1: Data Management}
\scriptsize

\textbf{A1: Airspace Loader}

\vspace{0.1cm}
Task: Loads the GraphML airspace network and checks waypoint locations and airway connections.

Tool: T1

\vspace{0.3cm}

\textbf{A2: Flight Data Checker}

\vspace{0.1cm}
Task: Matches aircraft specifications with flight plans, checks timestamps and prepares consistent Pydantic data.

Tool: T2

\end{block}

\end{column}

% RIGHT COLUMN
\begin{column}{0.49\textwidth}

\begin{block}{Crew 2: Surveillance \& Conflict}
\scriptsize

\textbf{A3: Aircraft Tracker}

\vspace{0.1cm}
Task: Predicts aircraft positions from the flight plans.

Tool: T3

\vspace{0.3cm}

\textbf{A4: Conflict Detector}

\vspace{0.1cm}
Task: Detects conflicts when horizontal separation is 5 NM or less AND vertical separation is 1,000 ft or less. Records CPA, time to loss of separation, conflict geometry, severity and ranking.

Tool: T4

\end{block}

\end{column}
