
% Slide 2: Problem and Goals

\begin{frame}{Problem and Goals}

\begin{block}{Problem}
\small
Aircraft are in \textbf{conflict} when they are predicted to be within
\textbf{5\,NM horizontally and 1000\,ft vertically} at the same moment.
Detecting, assessing and resolving conflicts needs distinct expertise,
so we model the task as \textbf{cooperating crews of agents}.
\end{block}

\begin{block}{System Goals}
\small
    \begin{itemize}
        \setlength{\itemsep}{1pt}
        \item Validate and normalise the input data.
        \item Detect, characterise and prioritise conflicts.
        \item Calculate and validate resolution manoeuvres.
        \item Generate flight-clearance reports.
    \end{itemize}
\end{block}

\begin{alertblock}{Scope}
\small
    Simulated batch system, not real-time ATC. Inputs: airspace (GraphML),
    aircraft and flight plans (JSON), performance limits, optional restrictions.
\end{alertblock}

\end{frame}
